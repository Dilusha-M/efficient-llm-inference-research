import math


def summarize_decode_speeds(lines):
    """
    Parse lines of key=value pairs and compute summary statistics of tps values
    from valid decode events.
    
    A record is valid if:
    - The line is a string
    - After splitting on whitespace, it contains both event=decode and tps=<value>
    - tps=<value> is a finite, strictly positive decimal accepted by float()
    - No duplicate keys with conflicting values (including duplicate tps with same value)
    
    Returns a dict with count, min_tps, max_tps, avg_tps.
    """
    count = 0
    min_tps = None
    max_tps = None
    total_tps = 0.0
    
    for line in lines:
        # Ignore non-string items
        if not isinstance(line, str):
            continue
        
        # Split on whitespace to get tokens
        tokens = line.split()
        
        # Parse key=value pairs
        # We need to track all keys and their values to detect duplicates
        kv_pairs = {}
        valid_kv = True
        
        for token in tokens:
            if '=' not in token:
                # Invalid token format, skip this record? 
                # Actually, the problem says "after splitting on whitespace, it contains both event=decode and a tps=<value> field"
                # So if there's a token without '=', we should still check for the required fields.
                # But malformed lines are ignored. A token without '=' could be part of a malformed line.
                # Let's be strict: each token should be key=value. If not, the line is malformed.
                valid_kv = False
                break
            
            key, _, value = token.partition('=')
            if key in kv_pairs:
                # Duplicate key
                # If same value, treat as malformed too
                if kv_pairs[key] != value:
                    valid_kv = False
                    break
                else:
                    # Same value duplicate, treat as malformed
                    valid_kv = False
                    break
            kv_pairs[key] = value
        
        if not valid_kv:
            continue
        
        # Check for event=decode
        if kv_pairs.get('event') != 'decode':
            continue
        
        # Check for tps
        if 'tps' not in kv_pairs:
            continue
        
        tps_str = kv_pairs['tps']
        
        # Parse tps value
        try:
            tps_val = float(tps_str)
        except (ValueError, OverflowError):
            continue
        
        # Check for finite and strictly positive
        if not math.isfinite(tps_val):
            continue
        if tps_val <= 0:
            continue
        
        # Valid record
        count += 1
        if min_tps is None or tps_val < min_tps:
            min_tps = tps_val
        if max_tps is None or tps_val > max_tps:
            max_tps = tps_val
        total_tps += tps_val
    
    if count == 0:
        avg_tps = None
    else:
        avg_tps = round(total_tps / count, 3)
    
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps,
    }
