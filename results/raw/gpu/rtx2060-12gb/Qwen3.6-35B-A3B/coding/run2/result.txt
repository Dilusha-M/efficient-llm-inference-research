import math


def summarize_decode_speeds(lines):
    """
    Summarize decode speeds from an iterable of log lines.
    
    A record is valid if it contains:
    - 'event=decode' key-value pair
    - 'tps=<value>' key-value pair where value is a finite, strictly positive float
    
    Returns a dict with count, min_tps, max_tps, avg_tps.
    """
    count = 0
    min_tps = None
    max_tps = None
    total_tps = 0.0
    
    for line in lines:
        # Ignore non-string items
        if not isinstance(line, str):
            continue
        
        # Split on whitespace to get tokens
        tokens = line.split()
        
        # Parse key=value pairs, detecting duplicates and conflicts
        kv_pairs = {}
        has_event_decode = False
        has_tps = False
        tps_value = None
        is_valid_record = True
        
        for token in tokens:
            if '=' not in token:
                continue
            
            # Split on first '=' only
            eq_idx = token.index('=')
            key = token[:eq_idx]
            value_str = token[eq_idx + 1:]
            
            if key == 'event' and value_str == 'decode':
                if key in kv_pairs and kv_pairs[key] != value_str:
                    # Duplicate key with conflicting value -> malformed
                    is_valid_record = False
                    break
                kv_pairs[key] = value_str
                has_event_decode = True
            elif key == 'tps':
                if key in kv_pairs:
                    # Duplicate tps key -> malformed (even if same value)
                    is_valid_record = False
                    break
                kv_pairs[key] = value_str
                has_tps = True
                # Try to parse the value
                try:
                    val = float(value_str)
                except (ValueError, OverflowError):
                    # Malformed float
                    is_valid_record = False
                    break
                # Check for non-finite, zero, or negative
                if not math.isfinite(val) or val <= 0:
                    is_valid_record = False
                    break
                tps_value = val
        
        if not is_valid_record:
            continue
        
        # Check that we have both required fields
        if not has_event_decode or not has_tps or tps_value is None:
            continue
        
        # Valid record
        count += 1
        total_tps += tps_value
        if min_tps is None or tps_value < min_tps:
            min_tps = tps_value
        if max_tps is None or tps_value > max_tps:
            max_tps = tps_value
    
    # Compute average
    if count == 0:
        avg_tps = None
    else:
        avg_tps = round(total_tps / count, 3)
    
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps,
    }
