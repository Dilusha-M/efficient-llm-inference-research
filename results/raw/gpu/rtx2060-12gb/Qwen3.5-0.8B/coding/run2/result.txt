from __future__ import annotations
from typing import Iterable, Optional

def summarize_decode_speeds(lines: Iterable[str]) -> dict[str, Optional[float]]:
    """
    Evaluate decoding speed from a list of lines (strings) using only the standard library.
    
    Returns a dictionary with keys:
    - 'count': number of valid decode records
    - 'min_tps': minimum valid tps, or None if no valid records
    - 'max_tps': maximum valid tps, or None if no valid records
    - 'avg_tps': arithmetic mean rounded to three decimal places, or None if no valid records
    
    A record is valid only when:
    - The line contains both "event=decode" and a "tps=<value>" field.
    
    Rules for invalid records:
    - Ignore non-string items in the input.
    - Ignore malformed lines (missing required fields).
    - Ignore non-decode events (e.g., "read", "log", etc.).
    - Ignore duplicate "tps" keys with conflicting values.
    - Ignore non-finite values: nan, inf, -inf.
    - Ignore zero and negative values for tps.
    - Round the mean to three decimal places.
    
    Time Complexity: O(n) where n is the number of lines.
    Space Complexity: O(1) extra space (excluding the input iterable).
    """
    result = {}
    tps_values: set[float] = set()
    valid_records: set[int] = set()
    
    for line in lines:
        # Check if the input is a string. If not, ignore it.
        if not isinstance(line, str):
            continue
        
        # Split on whitespace only.
        tokens = line.split()
        
        # Check for required fields.
        if "event" not in tokens or "tps" not in tokens:
            continue
        
        # Extract the tps value.
        # Must be a finite, strictly positive decimal.
        tps = float(tokens[1])
        if tps < 0 or tps < 1 or tps == float("nan") or tps == float("inf"):
            continue
        
        # Check for duplicate tps keys.
        if tps in tps_values:
            continue
        
        # Record is valid.
        result["count"] += 1
        result["min_tps"] = min(result["min_tps"], tps)
        result["max_tps"] = max(result["max_tps"], tps)
        tps_values.add(tps)
        valid_records.add(len(tokens))
    
    # Compute mean, rounding to three decimal places.
    if result["count"] > 0:
        avg_tps = round(result["count"] / result["count"] * 1000.0, 3)
        result["avg_tps"] = avg_tps
    else:
        result["avg_tps"] = None
    
    return result

#### 1. Why naive substring matching can be wrong
Naive substring matching would look for the string `"event=decode"` followed by `"tps=..."` directly in the line. This approach fails because:
- It might find `"event=decode"` and then skip the `tps=...` part if it assumes the next characters are part of the token, missing the actual `tps` field.
- It incorrectly accepts records where `tps` appears to be missing but is actually on the same line as an unrelated event (e.g., `event=decode\tps=1.5`), which would be considered a decode event but fail the value validation.

#### 2. Why malformed duplicate handling is required
If we simply search for `"event=decode"` and then assume any subsequent `tps` token is valid, we risk:
- Accepting a record where `tps` is `nan`, `inf`, or `0` (due to string splitting behavior).
