def summarize_decode_speeds(lines):
    """
    Summarize decode speeds from an iterable of log lines.
    
    Returns a dictionary with:
        - count: number of valid decode records
        - min_tps: minimum valid tps, or None if no valid records
        - max_tps: maximum valid tps, or None if no valid records
        - avg_tps: arithmetic mean rounded to 3 decimal places, or None if no valid records
    
    A record is valid only if:
        - It contains exactly one 'event=decode' field.
        - It contains exactly one 'tps=<value>' field where <value> is a finite, 
          strictly positive float.
        - There are no duplicate keys with conflicting values.
    
    Invalid inputs (ignored):
        - Lines without 'event=decode'.
        - Lines with 'event' != 'decode'.
        - Malformed floats (non-finite, zero, negative).
        - Duplicate keys (either conflicting or same value).
        - Non-string items in the iterable.
    """
    
    count = 0
    min_tps = None
    max_tps = None
    sum_tps = 0.0
    
    for line in lines:
        # Skip non-string items gracefully
        if not isinstance(line, str):
            continue
            
        tokens = line.split()
        if not tokens:
            continue
            
        # Dictionary to track key-value pairs for the current line
        # Using a dictionary ensures we catch duplicate keys naturally.
        parsed = {}
        valid_line = True
        
        for token in tokens:
            if '=' not in token:
                valid_line = False
                break
            
            key, _, value = token.partition('=')
            key = key.strip()
            value = value.strip()
            
            # Check for duplicate keys
            if key in parsed:
                # Duplicate key found
                # According to spec: "Treat a duplicate tps key with the same value as malformed too."
                # So any duplicate makes the line invalid.
                valid_line = False
                break
            
            parsed[key] = value
        
        if not valid_line:
            continue
            
        # Check required fields
        if 'event' not in parsed or parsed['event'] != 'decode':
            continue
            
        if 'tps' not in parsed:
            continue
            
        tps_str = parsed['tps']
        
        try:
            tps = float(tps_str)
        except ValueError:
            continue
        
        # Check for finite, strictly positive values
        if not math.isfinite(tps):
            continue
        if tps <= 0:
            continue
        
        # If we reached here, this is a valid record
        count += 1
        sum_tps += tps
        
        if min_tps is None or tps < min_tps:
            min_tps = tps
        if max_tps is None or tps > max_tps:
            max_tps = tps
    
    if count == 0:
        return {
            'count': 0,
            'min_tps': None,
            'max_tps': None,
            'avg_tps': None
        }
    
    avg_tps = round(mean_tps := sum_tps / count, 3)
    
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps
    }
