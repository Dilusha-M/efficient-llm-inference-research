```python
from typing import Iterable
import math
import re

def summarize_decode_speeds(lines: Iterable[str]) -> dict:
    """
    Analyze a list of strings to extract and summarize decode speed records.
    
    A valid record requires:
    - A 'count' field (integer)
    - An 'event=decode' tag
    - A 'tps=<value>' field where value is a finite, strictly positive float
    
    Returns a dict with keys: count, min_tps, max_tps, avg_tps.
    None is returned for tps fields if no valid records are found.
    """
    count = 0
    tps_values = []
    
    for raw_line in lines:
        # Skip non-string items gracefully
        if not isinstance(raw_line, str):
            continue
            
        # Split by whitespace; tokens are already stripped of surrounding whitespace
        tokens = raw_line.split()
        
        if not tokens:
            continue
            
        # Must contain exactly one 'count' token
        if tokens[0] != 'count':
            continue
            
        # Must contain exactly one 'event=decode' token
        if tokens[1] != 'event=decode':
            continue
            
        # Must contain exactly one 'tps=...' token
        if len(tokens) < 3:
            continue
            
        # Extract tps value
        tps_str = tokens[2]
        
        # Validate: must be a finite positive float
        # Must not be inf, -inf, nan, or zero
        if not math.isfinite(float(tps_str)) or float(tps_str) <= 0:
            continue
            
        # Parse float
        try:
            tps = float(tps_str)
        except ValueError:
            continue
            
        # Check for duplicates with same value
        if tps in tps_values:
            continue
            
        tps_values.append(tps)
        count += 1
        
    # Compute statistics
    if not tps_values:
        min_tps = None
        max_tps = None
        avg_tps = None
    else:
        min_tps = min(tps_values)
        max_tps = max(tps_values)
        avg_tps = round(sum(tps_values) / len(tps_values), 3)
        
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps
    }
```
