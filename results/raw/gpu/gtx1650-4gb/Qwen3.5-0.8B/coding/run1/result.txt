from typing import List, Any

def summarize_decode_speeds(lines: List[str]) -> dict:
    """
    Summarize decode speeds for a list of line strings.
    
    Algorithm (one-pass, O(1) extra space):
    1. Initialize counters: count = 0, min_tps = inf, max_tps = -inf, sum = 0.
    2. Iterate through `lines`. If `line` is not a string, skip it (ignore malformed item).
    3. Split the line into tokens by whitespace.
    4. Check for valid keys and values:
       - Must contain both `event=decode` and exactly one `tps=...` key.
       - `tps` must be a finite, strictly positive decimal.
       - Ignore duplicates of `tps` values.
    5. If a record is valid, increment `count`, update min/max/sum, and set `tps` to the value.
    6. If no valid records found, return `{'count': 0, 'min_tps': None, 'max_tps': None, 'avg_tps': None}`.
    """
    count = 0
    min_tps = float('inf')
    max_tps = float('-inf')
    total = 0.0

    for line in lines:
        # Skip non-string items
        if not isinstance(line, str):
            continue

        tokens = line.split()

        # Check required keys
        if 'event' not in tokens or 'event' not in tokens[0]:
            continue

        # Check for exactly one tps key
        if tokens[1] == 'tps':
            # Get the value
            tps_val = tokens[2]
            
            # Skip if value is None or empty string (though tokenization usually removes trailing whitespace)
            if not tps_val or tps_val.strip() == '':
                continue

            # Validate type: float only
            try:
                tps_val = float(tps_val)
            except (ValueError, TypeError):
                # Non-finite values are invalid
                continue

            # Validate strictly positive
            if tps_val <= 0:
                continue

            # Update counters (ignore duplicates of the same value)
            count += 1
            if tps_val < min_tps:
                min_tps = tps_val
            if tps_val > max_tps:
                max_tps = tps_val
            total += tps_val

    if count == 0:
        return {
            'count': 0,
            'min_tps': None,
            'max_tps': None,
            'avg_tps': None
        }
    
    avg_tps = total / count
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps
    }
