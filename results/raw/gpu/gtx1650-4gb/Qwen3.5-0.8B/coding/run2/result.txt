from typing import Iterable, Tuple, Any, Optional
import math

def summarize_decode_speeds(lines: Iterable[str]) -> Tuple[Optional[int], Optional[int], Optional[int], Optional[int]]:
    """
    Summarize decode speeds from a stream of lines.
    
    Input:
    lines: An iterable of strings. Non-string items are ignored.
    
    Output:
    Tuple of (count, min_tps, max_tps, avg_tps)
        where:
        - count: number of valid decode records
        - min_tps: minimum valid tps, or None if no valid records
        - max_tps: maximum valid tps, or None if no valid records
        - avg_tps: arithmetic mean rounded to three decimal places, or None if no valid records
    """
    # Track statistics across the entire input
    count = 0
    min_tps = float('inf')
    max_tps = float('-inf')
    sum_tps = 0.0

    # Map keys to their values for comparison
    # Keys are expected to be simple identifiers (strings)
    key_to_value = {}

    for line in lines:
        if not isinstance(line, str):
            continue  # Skip non-string items per spec

        # Split on whitespace. This normalizes spaces around '=' and '=' without spaces.
        tokens = line.split()

        # Check for required keys: event=decode and tps=<value>
        event = tokens[0]
        tps_value = None

        # Find the key-value pair
        for token in tokens[1:]:
            if token == 'tps=':
                tps_value = tokens[2]
                break

        if event != 'event=decode':
            continue  # Ignore non-decode events

        # Parse tps_value as a finite, strictly positive decimal
        try:
            tps = float(tps_value)
        except (ValueError, TypeError):
            continue  # Skip malformed floats

        # Validate tps is finite and strictly positive
        if tps <= 0 or math.isnan(tps) or math.isinf(tps):
            continue  # Skip nan, inf, or negative values

        # Store key-value mapping for duplicate detection
        if tps in key_to_value:
            # Duplicate tps key with conflicting value -> malformed
            continue
        key_to_value[tps] = tps

        # Update summary statistics
        count += 1
        sum_tps += tps
        min_tps = min(min_tps, tps)
        max_tps = max(max_tps, tps)

    # If there are no valid records, all results are None
    if count == 0:
        return (None, None, None, None)

    avg_tps = sum_tps / count
    # Round to three decimal places
    avg_tps = round(avg_tps, 3)

    return (count, min_tps, max_tps, avg_tps)
