from typing import Iterator, Dict
import math

def summarize_decode_speeds(lines: Iterator[str]) -> Dict[str, float | None]:
    """
    Summarize decode speed statistics from a generator of string lines.
    
    A valid record requires:
    - Event "decode" (exact match)
    - Key "tps" followed by a finite, strictly positive float
    - No duplicate "tps" keys with conflicting values
    
    Returns:
        {
            'count': int,
            'min_tps': float | None,
            'max_tps': float | None,
            'avg_tps': float | None
        }
    """
    if not lines:
        return {
            'count': 0,
            'min_tps': None,
            'max_tps': None,
            'avg_tps': None
        }
    
    seen_tps: Dict[float, bool] = {}  # keys: tps value, value: is seen
    valid_records = []  # stores (tps_value, count)
    min_tps = None
    max_tps = None
    count = 0
    
    for line in lines:
        # Skip non-string items gracefully
        if not isinstance(line, str):
            continue
            
        # Strip trailing whitespace just in case, though problem says surrounding is removed
        line = line.strip()
        
        if not line:
            continue
            
        # Split on whitespace
        parts = line.split()
        if not parts:
            continue
            
        # Check for event=decode
        event_parts = parts[0].split('=')
        if len(event_parts) != 2 or event_parts[0].strip() != 'event':
            continue
            
        # Check for tps=...
        tps_parts = parts[1].split('=')
        if len(tps_parts) != 2 or tps_parts[0].strip() != 'tps':
            continue
            
        tps_str = tps_parts[1].strip()
        
        # Parse float
        try:
            tps_value = float(tps_str)
        except ValueError:
            # Invalid float representation
            continue
            
        # Check constraints
        # 1. Strictly positive (not <= 0)
        if tps_value <= 0:
            continue
            
        # 2. Finite (no NaN, Inf)
        if not math.isfinite(tps_value):
            continue
            
        # 3. No zero
        if tps_value == 0.0:
            continue
            
        # 4. No negative
        if tps_value < 0:
            continue
            
        # Check for duplicate tps key with conflicting value
        if tps_value in seen_tps:
            # Conflict detected; skip this record
            continue
        else:
            seen_tps[tps_value] = True
            
        # Update stats
        count += 1
        if min_tps is None or tps_value < min_tps:
            min_tps = tps_value
        if max_tps is None or tps_value > max_tps:
            max_tps = tps_value
        valid_records.append(tps_value)
        
    # Compute average
    if count == 0:
        avg_tps = None
    else:
        avg_tps = sum(valid_records) / count
    
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps

### Explanation of Implementation

The `summarize_decode_speeds` function implements a one-pass algorithm with O(1) extra space (excluding input). It iterates through the input lines once, tracking:
- `seen_tps`: A dictionary to track which tps values have been encountered (to detect duplicates).
- `valid_records`: A list to accumulate valid tps values.
- `min_tps`, `max_tps`, `count`: State variables for computing statistics.

### Why Naive Substring Matching Fails

1. **Wrong Event**: If the parser incorrectly identifies the first token after `event=` as the event type instead of the field name, it might flag `event=other` as a valid decode event, leading to incorrect statistics.
2. **Malformed Duplicate**: A naive approach might skip a line with `tps=100` only if it doesn't find a matching `event=decode` elsewhere, but if it encounters `tps=100` in a different context or with a different event, it might incorrectly count it.
3. **Empty Result Policy**: The function returns `None` for `min_tps`, `max_tps`, and `avg_tps` when there are no valid records. This ensures the function doesn't crash and provides clear error conditions, rather than returning partial or incorrect statistics.
