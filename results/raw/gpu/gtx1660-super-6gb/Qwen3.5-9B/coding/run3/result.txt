def summarize_decode_speeds(lines: Iterable[str]) -> Dict[str, Any]:
    """
    Summarize decode speeds from a log-like input.
    
    Rules:
    - A record is valid only if it contains 'event=decode' AND 'tps=<value>'.
    - <value> must be a finite, strictly positive float.
    - Duplicate keys with conflicting values make the line malformed.
    - Duplicate keys with the SAME value are also considered malformed (per spec).
    - Ignore non-string items in the iterable.
    - Ignore lines with 'event' != 'decode' or missing 'tps'.
    - Ignore non-finite (nan, inf), zero, or negative tps.
    
    Returns:
        {
            'count': int,
            'min_tps': float or None,
            'max_tps': float or None,
            'avg_tps': float (rounded to 3 decimals) or None
        }
    """
    
    # One-pass state
    count = 0
    min_tps: Optional[float] = None
    max_tps: Optional[float] = None
    sum_tps = 0.0
    
    for line in lines:
        # Skip non-string items gracefully
        if not isinstance(line, str):
            continue
        
        # Tokenize by whitespace
        tokens = line.split()
        
        # We need to track keys seen in this line to detect duplicates
        seen_keys: Dict[str, str] = {}
        is_malformed = False
        
        for token in tokens:
            if '=' not in token:
                # Ignoring tokens without '=' is fine, they don't contribute to key conflicts
                # unless they are part of a malformed key=value pair, but the spec says 
                # "after splitting on whitespace, it contains both event=decode and a tps=<value>"
                # This implies we look for key=value tokens. If a token lacks '=', it's just ignored.
                continue
            
            key, _, value_str = token.partition('=')
            
            # Check for duplicate keys in the same line
            if key in seen_keys:
                # Spec: "Treat a duplicate tps key with the same value as malformed too."
                # This implies ANY duplicate key makes the line malformed.
                is_malformed = True
                break
            
            seen_keys[key] = value_str
        
        if is_malformed:
            continue
            
        # Check required fields
        if 'event' not in seen_keys or 'tps' not in seen_keys:
            continue
        
        event_val = seen_keys['event']
        tps_str = seen_keys['tps']
        
        # Check event type
        if event_val != 'decode':
            continue
        
        # Parse tps value
        try:
            tps_val = float(tps_str)
        except ValueError:
            continue
        
        # Check for non-finite values
        if isnan(tps_val) or isinf(tps_val):
            continue
        
        # Check strictly positive
        if tps_val <= 0:
            continue
        
        # Valid record found
        count += 1
        sum_tps += tps_val
        
        if min_tps is None or tps_val < min_tps:
            min_tps = tps_val
        if max_tps is None or tps_val > max_tps:
            max_tps = tps_val
            
        # Early exit optimization: if we have data, we can compute average at the end
        # But we must continue to process all lines for one-pass correctness.

    if count == 0:
        return {
            'count': 0,
            'min_tps': None,
            'max_tps': None,
            'avg_tps': None
        }
    
    avg_tps = sum_tps / count
    # Round to three decimal places
    avg_tps = round(avg_tps, 3)
    
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps
    }
