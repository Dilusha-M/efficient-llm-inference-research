import math


def summarize_decode_speeds(lines):
    """
    Process an iterable of strings to extract decode speeds (tps) from valid records.
    
    A valid record is a line that, when split on whitespace, contains exactly one
    'event=decode' token and exactly one 'tps=<value>' token where <value> is a
    finite, strictly positive float.
    
    Returns a dictionary with keys: count, min_tps, max_tps, avg_tps
    """
    count = 0
    min_tps = None
    max_tps = None
    total_tps = 0.0
    
    for line in lines:
        # Ignore non-string items
        if not isinstance(line, str):
            continue
        
        # Split on whitespace to get tokens
        tokens = line.split()
        
        # We need to find event=decode and tps=<value>
        # Check for duplicates and validity
        event_found = None
        tps_found = None
        tps_value = None
        
        for token in tokens:
            if '=' not in token:
                continue
            
            # Split on first '=' only
            key, _, value = token.partition('=')
            
            if key == 'event':
                if event_found is not None:
                    # Duplicate event key - malformed
                    event_found = None  # Mark as malformed
                    break
                event_found = value
            elif key == 'tps':
                if tps_found is not None:
                    # Duplicate tps key - malformed
                    tps_found = None  # Mark as malformed
                    break
                tps_found = value
                try:
                    val = float(value)
                except (ValueError, OverflowError):
                    tps_value = None
                    break
                # Check for finite and strictly positive
                if not math.isfinite(val) or val <= 0:
                    tps_value = None
                    break
                tps_value = val
        
        # Check if we have a valid record
        if event_found is None or tps_found is None or event_found != 'decode' or tps_value is None:
            continue
        
        # Valid record
        count += 1
        if min_tps is None or tps_value < min_tps:
            min_tps = tps_value
        if max_tps is None or tps_value > max_tps:
            max_tps = tps_value
        total_tps += tps_value
    
    if count == 0:
        return {
            'count': 0,
            'min_tps': None,
            'max_tps': None,
            'avg_tps': None
        }
    
    avg_tps = round(total_tps / count, 3)
    
    return {
        'count': count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps
    }
