import math

def summarize_decode_speeds(lines):
    """
    Summarizes decode speeds from an iterable of strings using a single pass.
    
    Complexity:
    - Time: O(N) where N is the number of characters in the input.
    - Space: O(1) extra space (storing only running metrics).
    """
    count = 0
    min_tps = float('inf')
    max_tps = float('-inf')
    total_tps = 0.0

    for line in lines:
        # Requirement: Ignore non-string items
        if not isinstance(line, str):
            continue

        tokens = line.split()
        
        event_found = False
        tps_value = None
        is_malformed = False
        
        # We must parse tokens to ensure we don't accept partial matches 
        # (e.g., 'event=decode_extra' or 'tps=1.0 tps=2.0')
        for token in tokens:
            if '=' not in token:
                continue
            
            key, val = token.split('=', 1)
            
            if key == 'event':
                if val == 'decode':
                    event_found = True
                else:
                    # If event is present but not 'decode', this line is not a decode record
                    # We don't break yet because we need to check for duplicate key violations
                    pass
            
            elif key == 'tps':
                try:
                    current_val = float(val)
                    # Requirement: strictly positive, finite, and not NaN/Inf
                    if not (math.isfinite(current_val) and current_val > 0):
                        is_malformed = True
                        break
                    
                    if tps_value is not None:
                        # Requirement: Treat duplicate tps keys (even same value) as malformed
                        is_malformed = True
                        break
                    tps_value = current_val
                except ValueError:
                    is_malformed = True
                    break
            else:
                # Other keys are ignored as per requirements
                pass

        # Final validation for the record
        if not is_malformed and event_found and tps_value is not None:
            count += 1
            total_tps += tps_value
            if tps_value < min_tps:
                min_tps = tps_value
            if tps_value > max_tps:
                max_tps = tps_value

    # Empty-result policy: return None for numeric fields if no valid records found
    if count == 0:
        return {
            "count": 0,
            "min_tps": None,
            "max_tps": None,
            "avg_tps": None
        }

    return {
        "count": count,
        "min_tps": min_tps,
        "max_tps": max_tps,
        "avg_tps": round(total_tps / count, 3)
    }

### Technical Explanations

#### 1. Why naive substring matching is dangerous
Using `if "event=decode" in line:` is fragile for two reasons:
* **Incorrect Event Matching:** It would incorrectly validate a line like `event=decode_extra_metadata`, which is a different event type. By splitting into tokens and checking the exact key/value equality, we ensure we only match the specific event requested.
* **Malformed Duplicates:** A naive approach might find the first `tps=10.0` and stop. However, the requirements state that duplicate keys (e.g., `tps=10.0 tps=10.0`) make the record malformed. A substring approach ignores the presence of the second key, whereas a token-based approach allows us to detect and invalidate the record.

#### 2. The Empty-Result Policy
The requirement specifies returning `None` for `min_tps`, `max_tps`, and `avg_tps` when no valid records exist. This is a critical production pattern to distinguish between a "zero value" (which might be a valid speed in some contexts, though not here) and "no data available." It prevents the function from returning `inf` or crashing with a `ZeroDivisionError` when calculating the mean.
