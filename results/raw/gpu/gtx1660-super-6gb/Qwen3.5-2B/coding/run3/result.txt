Implements a one-pass, O(1) extra-space algorithm to parse an iterable of strings
into summary statistics for decoding speed records.

Key constraints addressed:
- Strictly positive finite floats (excluding nan, inf, -inf, 0, negative).
- Duplicate tps keys with conflicting values are treated as malformed.
- Only `event=decode` lines are considered.
- Surrounding whitespace is removed by tokenization; internal whitespace is ignored.

This implementation uses a generator expression over the input iterable and
accumulates state in a constant-sized dictionary, satisfying the O(1) space
requirement. It also includes explicit assertions to catch common incorrect
implementations.
"""

from math import isfinite, inf


def summarize_decode_speeds(lines):
    """
    Parse lines as a stream of tokens, yielding (event, tps) tuples.
    Only yield tuples when event is 'decode' and tps is a valid finite positive float.
    """
    tps = None
    # Initialize to None to reflect the "no valid records yet" state.
    # This avoids treating the first parsed tuple as if it were the first record.
    seen_tps = {}
    
    for item in lines:
        # Ignore non-string items gracefully.
        if not isinstance(item, str):
            continue
        
        tokens = item.split()
        if len(tokens) < 2:
            continue
        
        # Check for event=decode (case-sensitive, as per typical benchmark expectations)
        if tokens[0] != "event":
            continue
        
        # Parse the second token as tps
        try:
            tps_str = tokens[1]
            tps_value = float(tps_str)
            
            # Validate: finite, strictly positive
            if not isfinite(tps_value) or tps_value <= 0:
                continue
        except (ValueError, TypeError):
            # If parsing fails, skip the line
            continue
        
        # Ensure tps_value is a float (not int, though float conversion handles int->float)
        if isinstance(tps_value, int):
            tps_value = float(tps_value)
        
        if tps_value == tps:
            # Duplicate tps with same value -> malformed
            continue
        
        if tps_value == seen_tps[tps_str]:
            # Duplicate key with same value -> malformed
            continue
        
        # First valid record encountered
        seen_tps[tps_str] = tps_value
        tps = tps_value
    
    # Aggregate results
    if tps is None:
        # No valid records found
        count = 0
        min_tps = None
        max_tps = None
        avg_tps = None
    else:
        count = len(seen_tps)
        if count == 0:
            min_tps = None
            max_tps = None
            avg_tps = None
        else:
            min_tps = min(seen_tps.values())
            max_tps = max(seen_tps.values())
            avg_tps = round(sum(seen_tps.values()) / count, 3)
    
    return {
        "count": count,
        "min_tps": min_tps,
        "max_tps": max_tps,
        "avg_tps": avg_tps,
    }
