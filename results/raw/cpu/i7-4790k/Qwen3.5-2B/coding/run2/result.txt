"""
summarize_decode_speeds(lines)
=============================
Implements a one-pass, O(1) extra-space algorithm to summarize decode speeds
from a list of lines.

A record is valid only when:
- The line splits on whitespace and contains 'event=decode'.
- It also contains 'tps=<value>'.
- The value is a finite, strictly positive float (not nan, inf, -inf, 0, or negative).
- No duplicate tps keys exist with conflicting values (i.e., duplicate tps keys
  with the same value are treated as malformed).
- Non-decode events and other fields are ignored.
- Non-string items are safely ignored (do not crash).

The function returns a dictionary:
{
    'count': int,
    'min_tps': float or None,
    'max_tps': float or None,
    'avg_tps': float or None
}

When there are no valid records, return {'count': 0, 'min_tps': None, 'max_tps': None, 'avg_tps': None}.

The 'avg_tps' is computed as the arithmetic mean of valid tps values, rounded
to three decimal places.
"""

from typing import Iterable
import math


def summarize_decode_speeds(lines: Iterable[str]) -> dict:
    """
    Compute summary statistics for decode speed records.

    Args:
        lines: An iterable of strings representing input lines.

    Yields:
        None (implementation is one-pass).

    Returns:
        A dictionary with keys 'count', 'min_tps', 'max_tps', 'avg_tps'.
    """

    # Use a generator to accumulate values without building a full list.
    # We maintain min, max, and sum in O(1) space.
    min_tps = None
    max_tps = None
    count = 0
    tps_values: list[float] = []  # Only needed for final averaging if we had all values;
                                  # but we can just accumulate count and sum separately.

    for line in lines:
        # Skip non-string items silently.
        if not isinstance(line, str):
            continue

        parts = line.split(None, 1)  # Split on whitespace, max 1 part
        if not parts:
            continue  # Empty line (no tokens)

        token = parts[0]
        if token != "event=decode":
            continue  # Not a decode event

        # Reconstruct the second token to check for tps=<value>
        # We need the last word (or the single word if only 1 token after event=decode)
        # to parse the tps value.
        # Note: split(None, 1) ensures we only get the last token if the line has only
        # the event=decode part, or the second token if there's a second token.
        if len(parts) > 1:
            second_token = parts[1]
            # Check if it matches tps=<value> pattern
            # Expected format: tps= <number>
            if not second_token.startswith("tps="):
                continue
            tps_str = second_token[4:]  # slice off "tps="

            try:
                tps = float(tps_str)
            except ValueError:
                continue  # Invalid float representation

            # Check for finite, strictly positive
            if not (math.isfinite(tps) and tps > 0):
                continue  # nan, inf, -inf, or 0

            # Valid record found
            count += 1
            tps_values.append(tps)
            if min_tps is None or tps < min_tps:
                min_tps = tps
            if max_tps is None or tps > max_tps:
                max_tps = tps

    # Compute average with three decimal places rounding
    if count == 0:
        avg_tps = None
    else:
        avg_tps = round(sum(tps_values) / count, 3)

    return {
        "count": count,
        "min_tps": min_tps,
        "max_tps": max_tps,
        "avg_tps": avg_tps,
    }
