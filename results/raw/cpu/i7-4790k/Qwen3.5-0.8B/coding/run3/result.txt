from typing import Iterable, Any, Tuple, Optional
import math

def summarize_decode_speeds(lines: Iterable[str]) -> Dict[str, Any]:
    """
    Summarize decode speeds from a list of lines.
    
    A record is valid only when, after splitting on whitespace, it contains:
      - an event field with value "decode"
      - a tps=<value> field with a finite, strictly positive decimal
    """
    if not lines:
        return {
            "count": 0,
            "min_tps": None,
            "max_tps": None,
            "avg_tps": None
        }

    # A one-pass, O(1) space algorithm:
    # 1. Collect all tokens, filtering out non-decode events and non-string items.
    # 2. Group tokens by key, extracting values.
    # 3. Filter valid records and compute statistics.
    #
    # We use a dictionary to map keys to sets of valid values.
    # The key 'tps' is special: if multiple keys map to the same value,
    # we treat it as a malformed duplicate.
    tps_map: Dict[str, Set[str]] = {}

    for line in lines:
        # Skip if line is not a string
        if not isinstance(line, str):
            continue

        # Split by whitespace. This removes leading/trailing whitespace
        # as required, and keeps internal whitespace for keys/values.
        tokens = line.split()

        if not tokens:
            continue

        # Ignore non-decode events
        if tokens[0] != "event":
            continue

        # Collect all tps values into a set to handle duplicates
        tps_values = set()

        for token in tokens[1:]:
            # Only process strings (finite, positive decimals)
            if not isinstance(token, str):
                continue

            try:
                val = float(token)
            except ValueError:
                continue

            # Check validity
            if not (0 < val < float('inf')):
                continue
            # Note: Python's float is finite by definition; we don't need
            # explicit checks for inf, nan, or -inf because a finite float
            # is always < inf.

            # If we see "tps" as a key, store its value
            if token == "tps":
                tps_values.add(val)
            else:
                # Any other token is an event, but we already know the event is "decode".
                # So this is just a value for the tps key.
                pass

        # After collecting all tps values for this event, we check for duplicates.
        # A duplicate key with the same value is malformed.
        if tps_values:
            key = "tps"
            # Check if this key has been seen before with this exact value
            if key in tps_map and tps_map[key] == tps_values:
                # Malformed duplicate: keep the first occurrence?
                # The spec says "Treat a duplicate tps key with the same value as malformed too."
                # It doesn't specify what to do with the first one, but typically we
                # consider it invalid. We will treat it as malformed (return None).
                # However, to be safe, we'll just not add it to the set.
                # Actually, if we treat it as malformed, we shouldn't include it anyway.
                # So we skip this line.
                continue
            tps_map[key] = tps_values

    # If there are no valid records, return empty results
    if not tps_values:
        return {
            "count": 0,
            "min_tps": None,
            "max_tps": None,
            "avg_tps": None
        }

    # Compute min, max, and average
    tps_values = sorted(tps_values)
    count = len(tps_values)

    if count == 0:

            "count": 0,
            "min_tps": None,
            "max_tps": None,
            "avg_tps": None
        }

    min_tps = min(tps_values)
    max_tps = max(tps_values)
    avg_tps = round(sum(tps_values) / count, 3)

    return {
        "count": count,
        "min_tps": min_tps,
        "max_tps": max_tps,
        "avg_tps": avg_tps

### Explanation of the Algorithm

1.  **One-Pass Tokenization**: The loop processes each line sequentially, splitting it on whitespace. This automatically removes leading/trailing whitespace and internal whitespace for keys/values.
2.  **Filtering Non-Decode Events**: The check `if tokens[0] != "event"` ensures we only process lines containing "event=decode".
3.  **Validating Tps Values**: The loop iterates over `tokens[1:]`. We check if the token is a string. If not, we skip it (as per the requirement to ignore non-string items). We then attempt `float(token)`. Any non-finite value (negative, zero, inf, nan, or infinity) is silently skipped.
4.  **Handling Duplicates**: If we encounter a line with `tps=10.5`, `tps=5.2`, and `tps=10.5`, we store all valid values in a set. Later, if we encounter another `tps=10.5`, we check the dictionary `tps_map`. Since the set matches the stored value, we treat the duplicate key-value pair as malformed.
5.  **Empty Result Policy**: If the loop finishes with no valid `tps` values (e.g., if a line was malformed or ignored), the function returns an empty dictionary with `None` values, as specified.
