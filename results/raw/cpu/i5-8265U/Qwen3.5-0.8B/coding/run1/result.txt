summarize_decode_speeds(lines):
    Process an iterable of strings, returning a summary dictionary of decode record statistics.

    Rules:
    - Split each line on whitespace to extract tokens.
    - Check that exactly two tokens are present: `event=decode` and `tps=<value>`.
    - Extract the numeric value for tps (must be finite, strictly positive decimal).
    - Ignore lines that do not match the required structure.
    - Ignore lines that are not strings.
    - Ignore duplicate `tps` keys with the same value.
    - Ignore malformed floats (`nan`, `inf`, `-inf`, zero, negative values).
    - Ignore non-string input items.

    Return dict with keys:
        - 'count': number of valid records
        - 'min_tps': min valid tps value, or None if no valid records
        - 'max_tps': max valid tps value, or None if no valid records
        - 'avg_tps': arithmetic mean rounded to 3 decimal places, or None if no valid records
"""

from typing import Iterable, Optional

def summarize_decode_speeds(lines: Iterable[str]) -> dict:
    """
    Process an iterable of strings and return statistics on decode records.

    Implementation details:
    - Use a set to track seen duplicate tps keys.
    - Maintain a running sum for valid tps values.
    - Use a set to track seen events to ensure we don't double-count or miss non-decode lines.
    - Use a set to track already seen strings to ignore non-string items.

    Time complexity: O(n), where n is the total number of tokens in the iterable.
    Space complexity: O(1) auxiliary space (excluding input iterable).
    """

    seen_duplicates: set[str] = set()   # {tps_key: value}
    seen_events: set[str] = set()       # set of event strings
    seen_strings: set[str] = set()      # set of non-string strings
    valid_tps_values: set[float] = set() # {value} for valid tps values
    total_tps_sum: float = 0.0
    valid_count: int = 0

    for line in lines:
        # Skip non-string items
        if not isinstance(line, str):
            continue

        # Split on whitespace
        tokens = line.split()

        # Must have exactly two tokens: 'event=decode' and 'tps=<value>'
        if len(tokens) != 2:
            continue

        event = tokens[0]
        tps_value_str = tokens[1]

        # Check for a decode event
        if event != "decode":
            continue

        # Check for a tps key with a finite, strictly positive decimal
        try:
            tps_value = float(tps_value_str)
            if tps_value <= 0 or not isinstance(tps_value, (int, float)):
                continue
        except (ValueError, TypeError):
            continue

        # Check for duplicate tps key with same value
        if tps_value in seen_duplicates:
            continue

        # Add to seen duplicates
        seen_duplicates.add(tps_value)

        # Add to seen events
        seen_events.add(event)

        # Add to seen strings (non-decode events, non-decode lines, etc.)
        seen_strings.add(tps_value_str)

        # Add to valid tps values
        valid_tps_values.add(tps_value)

        # Accumulate tps sum
        total_tps_sum += tps_value

        # Increment valid count
        valid_count += 1

    # Compute summary statistics
    if valid_count == 0:
        return {
            'count': 0,
            'min_tps': None,
            'max_tps': None,
            'avg_tps': None
        }

    # min/max/mean
    if valid_tps_values:
        min_tps = min(valid_tps_values)
        max_tps = max(valid_tps_values)
        avg_tps = round(total_tps_sum / valid_count, 3)
    else:
        min_tps = max_tps = avg_tps = None

    return {
        'count': valid_count,
        'min_tps': min_tps,
        'max_tps': max_tps,
        'avg_tps': avg_tps
    }
